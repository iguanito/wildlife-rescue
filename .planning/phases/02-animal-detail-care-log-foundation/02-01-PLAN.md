  <action>
    Add to api/routes/animals.js — insert these two routes immediately before module.exports, after the existing POST /:id/medical route.

    Step 1: Add require at top of file (after existing requires):
    const CareLog = require('../models/CareLog');

    Step 2: Add GET route:
    ```javascript
    // GET /api/animals/:id/carelogs
    router.get('/:id/carelogs', async (req, res) => {
      try {
        const logs = await CareLog.find({ animal: req.params.id })
          .sort({ date: -1 })
          .populate('createdBy', 'email role');
        res.json(logs);
      } catch (err) {
        res.status(500).json({ error: err.message });
      }
    });
    ```

    Step 3: Add POST route (per D-14: requireRole(['staff', 'vet', 'admin'])):
    ```javascript
    // POST /api/animals/:id/carelogs
    router.post('/:id/carelogs', requireRole('staff', 'vet', 'admin'), async (req, res) => {
      try {
        const log = await CareLog.create({
          ...req.body,
          animal: req.params.id,
          createdBy: req.user._id,
        });
        const populated = await CareLog.findById(log._id).populate('createdBy', 'email role');
        res.status(201).json(populated);
      } catch (err) {
        res.status(400).json({ error: err.message });
      }
    });
    ```

    CRITICAL: createdBy is always set from req.user._id. If req.body contains createdBy, the spread + override ensures it is ignored.
    CRITICAL: Use CareLog.findById().populate() for the POST response, not log.populate() — avoids Mongoose v8 method chaining issues per Phase 1 decision context.
    NOTE: The GET /:id/medical route does NOT currently populate createdBy. Do not change that route — it will be addressed separately. Only add the new carelogs routes.
